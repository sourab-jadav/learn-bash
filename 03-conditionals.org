# Conditionals

we can check things internal to bash

#+begin_src bash
a=2
b=2
if [[ $a == $b ]]; then
    echo they are the same
fi
#+end_src


we can check the outside world

#+begin_src bash
if [[ -f /etc/passwd ]]; then
    echo /etc/passwd is a file
fi  
#+end_src



we also have `while` and `until`

#+begin_src bash
while [[ -f flag.txt ]]; do
    rm flag.txt
done 
echo done
#+end_src


#+begin_src bash
until [[ -f flag.txt ]]; do
    touch flag.txt
done
echo done
#+end_src
