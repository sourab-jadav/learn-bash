#+begin_src awk
  awk -F: '/fpr/ {print $10; exit}' data.txt 
                                    # in the file data.txt
                                    # traverse line by line
                                    # look for the line which has fpr in it
                                    # : is field seperator
                                    # field value $(value) start from 1
                                    # That's it - first line containing fpr, 10th colon-separated field, then stop.
#+end_src

#+begin_src bash
  ~/work/learn-bash/awk/practise [2] $ awk '{print}' data.txt 
  john 25 engineer 5000
  mary 30 designer 6000
  alex 22 student 2000
  lisa 28 engineer 5500

  ~/work/learn-bash/awk/practise $ # print columns
  ~/work/learn-bash/awk/practise $ awk '{print $2}' data.txt 
  25
  30
  22
  28

  ~/work/learn-bash/awk/practise [2] $ awk '{print $NF}' data.txt # NF is number of fields
  5000
  6000
  2000
  5500

  ~/work/learn-bash/awk/practise $ awk '{print $NR}' data.txt # NR is Record number
  john
  30
  student
  5500

  ~/work/learn-bash/awk/practise $ awk '$2 >25' data.txt 
  mary 30 designer 6000
  lisa 28 engineer 5500

  ~/work/learn-bash/awk/practise $ awk '$2 > 25 {print $1}' data.txt 
  mary
  lisa

  ~/work/learn-bash/awk/practise [2] $ awk '$3=="engineer" && $4>5200' data.txt 
  lisa 28 engineer 5500

  ~/work/learn-bash/awk/practise [1] $ awk '$3 == "engineer" {count++} END {print count}' data.txt 
  2

  ~/work/learn-bash/awk/practise $ awk '$4>max max=$4 END {print $4}' data.txt # syntax error
  awk: line 1: syntax error at or near =

  ~/work/learn-bash/awk/practise [2] $ awk '$4>max {max=$4} END {print max}' data.txt 
  6000
  ~/work/learn-bash/awk/practise $ awk '{if($4 > max) max=$4} END { print max}' data.txt 
  6000

  awk -F: '{print $1, $7}' /etc/passwd # here ':' is the field seperator

  ~/work/learn-bash/awk/practise $ awk '$1 >25 && $NF > 3000 {print}' data.txt # multiple conditions
