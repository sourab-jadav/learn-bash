*** capture the last argument of the previous command
*!$*
#+begin_example

#+end_example


*** likewise getting other arguments
=!*=   *-all args of previous command*
=!:1=  *-first argument of prev command*
=!!=   *-entire previous argument*

#+end_src

*** rm
rm uses glob pattern for expansion and it does not use regular expressions
**** common usecases
- delete all .txt files
  rm *.txt
- delete files staring with log
  rm log*
- delete files ending with digit
  rm *[0-9]
- delete files like file1 file2 but not file10
  rm file?
- delete files except .c files
  rm !(*.c)
**** extended globing
enable it using
shopt -s extglob

***** if you want regex use find command
find . -type f -regex '.*[0-9]\.log'
find . -type f -regex '.*[0-9]\.log' -delete

***** always preview before deleting
ls *.log
find . -regex 'pattern'

***** interactive delete
rm -i *.log
**** by creating an alias
alias rm='rm -i'
rm lesson*
***** you can check an alias using :- alias rm

*** find cmd
**** imp attributes
- limit the level of search
  -maxdepth <num>
- type of file search
  -type d (or) f
- if any permission requirements
  -perm 700
**** use cases
***** list all the dir that has perm 700(private-only user can access)
#+begin_src bash
  find . -maxdepth 1 -type d -perm 700
#+end_src

*** type command
to check if  a command is bash builtin type of external command
- it will also list all types of aliases associated with the command
- for builtin commands you can use *help* command to display their man pages
- e.g: help echo
- help help
  
#+begin_src bash
  type history
  history is a shell builtin

  type -a ls                      # to list all the aliases
  type -a echo
  type -a kill
#+end_src

*** builtin variables in linux:
whenever you execute a command it looks in the path variable which has listing of all the paths that it needs to look
it searches for that cmd in those paths if the command found it executes it
#+begin_src bash
  echo $PATH
  echo $PWD
  echo $USER
  whoami
  type whoami
  echo $SHELL
  echo $HOSTNAME
  echo $MACHTYPE                  # tells about your machine
  # setting a variable
  foo='hello      world'
  echo foo
  hello world                     # hence always use "" to avoid is behaviour
  echo "$foo"
  hello     world
  unset foo                       # unsetting the variable
  echo $PATH | tr '\n'            # tr cmd is used as a pipe to replace some char of a stream to another char '\n' in this case
#+end_src

** bash syntax
*** storing a command output in a variable
#+begin_src bash
  thing = `uname -a`
  echo $thing                     # prints the same output as above
  # FYI don't use `` use $() as shown below
  thing = $(uname -a)
  echo "$thing"                   # prints the same output
#+end_src

#+begin_src bash
  #!/usr/bin/env bash

  name='dave'
  number=12
  system=$(uname)

  echo "hello $name (your number is $number)"
  echo "your system is : $system"
#+end_src
